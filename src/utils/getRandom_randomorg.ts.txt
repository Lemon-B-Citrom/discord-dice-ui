export default function getRandom(maxValue: number, minValue: number = 1) {
  console.log("Generating RNG for (" + minValue + ", " + maxValue + ")");
  const min = Math.ceil(minValue);
  const max = Math.floor(maxValue);
  
  return Math.floor(getSeed() * (max - min + 1)) + min;
}

export function getSimpleRandom(maxValue: number, minValue: number = 1) {
  const min = Math.ceil(minValue);
  const max = Math.floor(maxValue);
  
  return Math.floor(Math.random() * (max - min + 1)) + min;
}

let seeds: Array<number>;
seeds = [];

function populateSeeds()
{
    console.log("Populating RNG");
    const restReq = new XMLHttpRequest();
    restReq.open("POST", "https://api.random.org/json-rpc/4/invoke");
    restReq.setRequestHeader("Content-Type", "application/json; charset=UTF-8");

    const body = JSON.stringify({
        jsonrpc: "2.0",
        method: "generateDecimalFractions",
        params: {
            apiKey: "38e6a1d8-669a-4b54-88d7-d4cda7bd23cb",
            n: 100,
            decimalPlaces: 5,
            replacement: true
        },
        id: 40818
    });

    restReq.onload = () => {
      if (restReq.readyState == 4 && restReq.status == 200) {
        let resp = JSON.parse(restReq.responseText);
        seeds = seeds.concat(resp.result.random.data);
        console.log("Remaining requests: " + resp.result.requestsLeft);
      } else {
        console.log(`Error: ${restReq.status}`);
      }
    };

    restReq.send(body);
}

function getSeed(): number
{
  let seed: number;
  seed = seeds.pop() as number;
  if (seeds.length < 1)
  {
    console.log("Populating from getSeed(), length = " + seeds.length);
    populateSeeds();
  }
  return seed;
}

if (seeds.length < 1)
{
    console.log("Populating from initial");
    populateSeeds();
}
